cmake_minimum_required(VERSION 3.18)
project(lattice-engine-runtime LANGUAGES C CXX)

if(NOT LATTICE_LLAMA_SOURCE)
    message(FATAL_ERROR "Set LATTICE_LLAMA_SOURCE to the pinned llama.cpp checkout")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_RPC ON CACHE BOOL "" FORCE)
set(GGML_RPC_RDMA OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)

add_subdirectory("${LATTICE_LLAMA_SOURCE}" llama.cpp)

file(READ "${LATTICE_LLAMA_SOURCE}/common/arg.cpp" LATTICE_ARG_SOURCE)
string(REGEX MATCHALL "common_params_apply_system_config\\(params, ctx_arg\\.ex\\)" LATTICE_CONFIG_CALLS "${LATTICE_ARG_SOURCE}")
list(LENGTH LATTICE_CONFIG_CALLS LATTICE_CONFIG_CALL_COUNT)
if(NOT LATTICE_CONFIG_CALL_COUNT EQUAL 1)
    message(FATAL_ERROR "Expected exactly one llama.cpp system configuration call")
endif()
string(REPLACE "common_params_apply_system_config(params, ctx_arg.ex);" "" LATTICE_ARG_SOURCE "${LATTICE_ARG_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/lattice-arg.cpp.in" "${LATTICE_ARG_SOURCE}")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/lattice-arg.cpp.in" "${CMAKE_CURRENT_BINARY_DIR}/lattice-arg.cpp" COPYONLY)
set_source_files_properties("${LATTICE_LLAMA_SOURCE}/common/arg.cpp" TARGET_DIRECTORY llama-common PROPERTIES HEADER_FILE_ONLY TRUE)
target_sources(llama-common PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/lattice-arg.cpp")

add_executable(lattice-engine-info engine-info.cpp)
target_compile_features(lattice-engine-info PRIVATE cxx_std_17)
target_link_libraries(lattice-engine-info PRIVATE ggml)
set_target_properties(lattice-engine-info PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
